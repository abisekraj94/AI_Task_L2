Postgresql SQL table schemas to create the tables for this application

-- USERS TABLE
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(255) UNIQUE NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    hashed_password VARCHAR(255) NOT NULL,
    role VARCHAR(50) NOT NULL DEFAULT 'employee', -- employee, manager, HR
    is_active BOOLEAN DEFAULT TRUE
);

-- EMPLOYEES TABLE
CREATE TABLE employees (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    role VARCHAR(255) NOT NULL,
    department VARCHAR(255),
    manager_id INTEGER REFERENCES employees(id) ON DELETE SET NULL
);

-- GOALS TABLE
CREATE TABLE goals (
    id SERIAL PRIMARY KEY,
    employee_id INTEGER NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
    title VARCHAR(255) NOT NULL,
    description TEXT,
    start_date DATE,
    end_date DATE,
    status VARCHAR(50),
    progress FLOAT DEFAULT 0.0
);

-- REVIEWS TABLE
CREATE TABLE reviews (
    id SERIAL PRIMARY KEY,
    employee_id INTEGER NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
    reviewer_id INTEGER NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
    period VARCHAR(100),
    feedback TEXT,
    rating FLOAT,
    type VARCHAR(50) -- self, peer, manager
);

-- SKILLS TABLE
CREATE TABLE skills (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) UNIQUE NOT NULL,
    description TEXT
);

-- EMPLOYEE_SKILLS TABLE
CREATE TABLE employee_skills (
    id SERIAL PRIMARY KEY,
    employee_id INTEGER NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
    skill_id INTEGER NOT NULL REFERENCES skills(id) ON DELETE CASCADE,
    level VARCHAR(50),
    last_assessed DATE
);
